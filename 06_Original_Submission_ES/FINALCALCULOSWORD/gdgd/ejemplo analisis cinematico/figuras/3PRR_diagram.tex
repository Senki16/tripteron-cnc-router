\newcommand {\actuator}[5]{

\coordinate [] (b1) at ($(#1) +(\myangle+90:\tick*#3)$);
\coordinate [] (b2) at ($(#2) +(\myangle+90:\tick*#3)$);

\newcommand {\groundlink}[5]{
%\groundlink{p1}{angle}{width}{% of height}{color};
\begin{scope}[rotate=#2]
\coordinate (a1) at ($(#1)+(#3,0)$);
\coordinate (a2) at ($(#1)+(-1.5*#3,-#3*#4)$);
\draw [fill=#5](a1) arc (0:180:#3) -- ++(0,-#3*#4) -- ++(2*#3,0) -- (a1);
\draw [fill=white] (#1) circle (0.4*#3);
\draw [pattern=north west lines] (a2) rectangle +(3*#3,-0.4*#3);
\end{scope} }

\newcommand {\link}[4]{
%\link{p1}{p2}{width}{color};

\coordinate [] (a1) at ($(#1) +(\myangle+90:#3)$);
\coordinate [] (a2) at ($(#2) +(\myangle+90:#3)$);
\coordinate [] (a3) at ($(#1) +(\myangle-90:#3)$);
\draw [fill=#4] (a1) -- (a2) arc (\myangle+90:\myangle+90-180:#3) -- (a3) arc (\myangle-90:\myangle-90-180:#3) ;
\draw [fill=white] (#1) circle (0.4*#3);
\draw [fill=white] (#2) circle (0.4*#3);}
